### 2. Mock Engine Implementation
```typescript
// packages/cli/src/scenarios/mock-engine.ts
export class MockEngine {
  private mocks: MockDefinition[] = [];

  addMock(mock: MockDefinition) {
    this.mocks.push(mock);
  }

  async findMock(service: string, method: string, args: any[]): Promise<any> {
    const candidates = this.mocks.filter(m => 
      m.service === service && m.method === method
    );

    for (const mock of candidates) {
      if (await this.matchesCondition(mock, args)) {
        this.logger.info(`Mock triggered: ${service}.${method} with condition: ${JSON.stringify(mock.when)}`);
        return this.generateResponse(mock, args);
      }
    }

    return null; // No mock found
  }

  private async matchesCondition(mock: MockDefinition, args: any[]): Promise<boolean> {
    if (!mock.when) return true; // Default mock

    // Match on arguments
    if (mock.when.args) {
      if (!this.deepEqual(args, mock.when.args)) return false;
    }

    // Match on input parameters
    if (mock.when.input) {
      const input = this.extractInputFromArgs(args);
      if (!this.deepEqual(input, mock.when.input)) return false;
    }

    // Custom matcher function
    if (mock.when.matcher) {
      const matcherFn = new Function('args', 'input', mock.when.matcher);
      return matcherFn(args, this.extractInputFromArgs(args));
    }

    return true;
  }

  private generateResponse(mock: MockDefinition, args: any[]): any {
    if (mock.error) {
      throw new Error(`${mock.error.code}: ${mock.error.message}`);
    }

    if (mock.responseFn) {
      const responseFn = new Function('args', 'input', mock.responseFn);
      return responseFn(args, this.extractInputFromArgs(args));
    }

    return mock.response;
  }
}
```

        endTime,

import type { Argv } from 'yargs';
import yaml from 'js-yaml';
import fs from 'fs';
import path from 'path';
import { elizaLogger } from '@elizaos/core'; // Assuming a shared logger
// Define the arguments for the 'run' subcommand
interface ScenarioRunArgs {
  filePath: string;
  live: boolean;
}
// Command definition for 'scenario run'
export const command = 'scenario <command>';
export const desc = 'Manage and execute ElizaOS scenarios.';
export const builder = (yargs: Argv): Argv => {
  return yargs.command(
    'run <filePath>',
    'Execute a scenario from a YAML file.',
    (yargs: Argv) => {
      return yargs
        .positional('filePath', {
          describe: 'Path to the .scenario.yaml file',
          type: 'string',
          demandOption: true,
        })
        .option('live', {
          alias: 'l',
          type: 'boolean',
          description: 'Run scenario in live mode, ignoring mocks.',
          default: false,
        });
    },
    (argv: ScenarioRunArgs) => {
      // This is the handler function that executes when the command is called
      handleRunScenario(argv);
    }
  );
};
// The core logic for the command
function handleRunScenario(args: ScenarioRunArgs) {
  elizaLogger.info(`Starting scenario run with args: ${JSON.stringify(args)}`);
  try {
    const fullPath = path.resolve(args.filePath);
    elizaLogger.info(`Attempting to read scenario file from: ${fullPath}`);
    if (!fs.existsSync(fullPath)) {
      elizaLogger.error(`Error: File not found at '${fullPath}'`);
      process.exit(1);
    }
    const fileContents = fs.readFileSync(fullPath, 'utf8');
    const scenario = yaml.load(fileContents);
    console.log('--- Parsed Scenario Content ---');
    console.log(JSON.stringify(scenario, null, 2));
    console.log('-----------------------------');
    
    elizaLogger.info('Scenario file parsed successfully.');
  } catch (error) {
    elizaLogger.error('An error occurred during scenario execution:', error);
    process.exit(1);
  }
}

### 2. Mock Engine Implementation
```typescript
// packages/cli/src/scenarios/mock-engine.ts
export class MockEngine {
  private mocks: MockDefinition[] = [];

  addMock(mock: MockDefinition) {
    this.mocks.push(mock);
  }

  async findMock(service: string, method: string, args: any[]): Promise<any> {
    const candidates = this.mocks.filter(m => 
      m.service === service && m.method === method
    );

    for (const mock of candidates) {
      if (await this.matchesCondition(mock, args)) {
        this.logger.info(`Mock triggered: ${service}.${method} with condition: ${JSON.stringify(mock.when)}`);
        return this.generateResponse(mock, args);
      }
    }

    return null; // No mock found
  }

  private async matchesCondition(mock: MockDefinition, args: any[]): Promise<boolean> {
    if (!mock.when) return true; // Default mock

    // Match on arguments
    if (mock.when.args) {
      if (!this.deepEqual(args, mock.when.args)) return false;
    }

    // Match on input parameters
    if (mock.when.input) {
      const input = this.extractInputFromArgs(args);
      if (!this.deepEqual(input, mock.when.input)) return false;
    }

    // Custom matcher function
    if (mock.when.matcher) {
      const matcherFn = new Function('args', 'input', mock.when.matcher);
      return matcherFn(args, this.extractInputFromArgs(args));
    }

    return true;
  }

  private generateResponse(mock: MockDefinition, args: any[]): any {
    if (mock.error) {
      throw new Error(`${mock.error.code}: ${mock.error.message}`);
    }

    if (mock.responseFn) {
      const responseFn = new Function('args', 'input', mock.responseFn);
      return responseFn(args, this.extractInputFromArgs(args));
    }

    return mock.response;
  }
}

    // Create message for agent
    const message: Memory = {
      entityId: context?.user_id || 'scenario-user',
      roomId: context?.room_id || 'scenario-room',
      content: {
        type: 'text',
        text: input,
      },
      metadata: {
        type: 'message',
        conversationId: context?.conversation_id,
      },
    };

    // Send to agent and capture response
    const startTime = Date.now();

      text: response.text,
      thoughts: response.thoughts,
      actions: response.actions,
      timing: {
        startTime,
        endTime,
        duration: endTime - startTime,
      },
      context: {

        messageId: message.id,
      },
    };

// packages/cli/src/commands/scenario.ts
import type { Argv } from 'yargs';
import yaml from 'js-yaml';
import fs from 'fs';
import path from 'path';
import { elizaLogger } from '@elizaos/core'; // Assuming a shared logger
// Define the arguments for the 'run' subcommand
interface ScenarioRunArgs {
  filePath: string;
  live: boolean;
}
// Command definition for 'scenario run'
export const command = 'scenario <command>';
export const desc = 'Manage and execute ElizaOS scenarios.';
export const builder = (yargs: Argv): Argv => {
  return yargs.command(
    'run <filePath>',
    'Execute a scenario from a YAML file.',
    (yargs: Argv) => {
      return yargs
        .positional('filePath', {
          describe: 'Path to the .scenario.yaml file',
          type: 'string',
          demandOption: true,
        })
        .option('live', {
          alias: 'l',
          type: 'boolean',
          description: 'Run scenario in live mode, ignoring mocks.',
          default: false,
        });
    },
    (argv: ScenarioRunArgs) => {
      // This is the handler function that executes when the command is called
      handleRunScenario(argv);
    }
  );
};
// The core logic for the command
function handleRunScenario(args: ScenarioRunArgs) {
  elizaLogger.info(`Starting scenario run with args: ${JSON.stringify(args)}`);
  try {
    const fullPath = path.resolve(args.filePath);
    elizaLogger.info(`Attempting to read scenario file from: ${fullPath}`);
    if (!fs.existsSync(fullPath)) {
      elizaLogger.error(`Error: File not found at '${fullPath}'`);
      process.exit(1);
    }
    const fileContents = fs.readFileSync(fullPath, 'utf8');
    const scenario = yaml.load(fileContents);
    console.log('--- Parsed Scenario Content ---');
    console.log(JSON.stringify(scenario, null, 2));
    console.log('-----------------------------');
    
    elizaLogger.info('Scenario file parsed successfully.');
  } catch (error) {
    elizaLogger.error('An error occurred during scenario execution:', error);
    process.exit(1);
  }
}
