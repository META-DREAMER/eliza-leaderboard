import puppeteer from 'puppeteer';
import fs from 'fs/promises';
import path from 'path';
export async function generatePdfFromHtml(htmlContent: string, outputPath: string): Promise<void> {
  const tempHtmlPath = path.join(path.dirname(outputPath), 'temp_report.html');
  await fs.writeFile(tempHtmlPath, htmlContent);
  const browser = await puppeteer.launch({ headless: true });
  const page = await browser.newPage();
  
  await page.goto(`file://${tempHtmlPath}`, { waitUntil: 'networkidle0' });
  // Optional: Wait for charts to finish animating
  await page.waitForTimeout(2000); 
  await page.pdf({
    path: outputPath,
    format: 'A4',
    printBackground: true,
    margin: { top: '20px', right: '20px', bottom: '20px', left: '20px' }
  });
  await browser.close();
  await fs.unlink(tempHtmlPath);
}

#### **2. Description**

To enable advanced, multi-variable testing of agent performance, we need a robust way to define a "matrix" of test configurations. This ticket covers the creation of the YAML configuration file that will drive the new Scenario Matrix Runner.

The goal is to design a schema that is both powerful and easy for developers to understand. This configuration will specify a base scenario and a set of parameters to vary, allowing the system to automatically generate and run all possible combinations.

This work is foundational for the entire Scenario Testing and Reporting Framework and does not include the logic for actually running the scenarios, only for defining and validating their configuration.

#### **3. Acceptance Criteria**

1. **YAML Schema Definition:**
   * A formal YAML schema is defined and documented. The configuration file (e.g., `matrix.scenario.yaml`) must support the following top-level keys:
     * `name` (string, required): A human-readable name for the test matrix (e.g., "GitHub Issue Prompt Robustness").
     * `description` (string, optional): A longer description of the test's goals.
     * `base_scenario` (string, required): The file path (relative to the project root) to the base `.scenario.yaml` file to be used in each run.
     * `runs_per_combination` (integer, optional, default: `1`): The number of times to execute each unique combination of parameters to check for consistency.

   * This file will contain exported TypeScript interfaces (`MatrixConfig`, `MatrixAxis`, etc.) that precisely match the defined YAML schema, ensuring strong type safety for the rest of the feature.

   * The `matrix-schema.ts` file will also contain an exported validation function, e.g., `validateMatrixConfig(config: unknown): MatrixConfig`.
   * This function will use a library like **Zod** to rigorously validate the parsed YAML object against the schema.
   * Validation errors must be user-friendly and clearly indicate the specific problem in the configuration file (e.g., `Validation Error: "matrix[1].values" must be a non-empty array.`).
4. **Documentation:**
   * A new documentation file, `packages/cli/src/commands/scenario/docs/matrix-testing.md`, will be created.

import puppeteer from 'puppeteer';
import fs from 'fs/promises';
import path from 'path';
export async function generatePdfFromHtml(htmlContent: string, outputPath: string): Promise<void> {
  const tempHtmlPath = path.join(path.dirname(outputPath), 'temp_report.html');
  await fs.writeFile(tempHtmlPath, htmlContent);
  const browser = await puppeteer.launch({ headless: true });
  const page = await browser.newPage();
  
  await page.goto(`file://${tempHtmlPath}`, { waitUntil: 'networkidle0' });
  // Optional: Wait for charts to finish animating
  await page.waitForTimeout(2000); 
  await page.pdf({
    path: outputPath,
    format: 'A4',
    printBackground: true,
    margin: { top: '20px', right: '20px', bottom: '20px', left: '20px' }
  });
  await browser.close();
  await fs.unlink(tempHtmlPath);
}
```
