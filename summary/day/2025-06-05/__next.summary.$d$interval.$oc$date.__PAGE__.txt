      };
    }

    const result = (await response.json()) as {

    if (!result.id) {
      logger.error("Workflow execution failed: Unknown error");

    const message = error instanceof Error ? error.message : String(error);

    return { success: false, error: `Error: ${message}` };
  }
}

/**
 * Polls the comfy.icu API for workflow completion
 */
async function pollForCompletion(
  runtime: IAgentRuntime,
  workflowId: string,
  runId: string
): Promise<string | null> {
  const apiKey = getApiKey(runtime);
  const maxAttempts = 30; // 5 minutes with 10-second intervals
  let attempts = 0;

  while (attempts < maxAttempts) {
    try {
      const response = await fetch(
        `https://comfy.icu/api/v1/workflows/${workflowId}/runs/${runId}`,
        {
          headers: {
            Authorization: `Bearer ${apiKey}`,
          },
        }
      );

      if (!response.ok) {
        logger.warn(`Status check failed: ${response.status}`);
        await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
        attempts++;
        continue;
      }

      const status = (await response.json()) as {
        status: "STARTED" | "COMPLETED" | "ERROR";
        output?: Array<{ url: string }>;
        error?: string;
      };

      if (
        status.status === "COMPLETED" &&
        status.output &&
        status.output.length > 0
      ) {
        // Get the first image URL
        const imageUrl = status.output[0].url;
        if (imageUrl) {
          logger.info(`Workflow completed, image URL: ${imageUrl}`);
          return imageUrl;
        } else {
          logger.error("Workflow completed but no image URL found");
          return null;
        }
      } else if (status.status === "ERROR") {
        logger.error(`Workflow failed: ${status.error || "Unknown error"}`);
        return null;
      }

      // Still processing, wait and try again
      logger.debug(`Workflow status: ${status.status}, waiting...`);
      await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
      attempts++;
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      logger.error(`Error checking workflow status: ${message}`);
      await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
      attempts++;
    }
  }

  logger.error("Timed out waiting for workflow completion");
  return null;
}

### Problem
The `ensureConnections` function was attempting to insert participants before ensuring the room exists, causing a foreign key constraint violation:

```
[2025-06-04 22:42:47] ERROR: Error adding participants
    entityIdSample: "b850bc30-45f8-0041-a00a-83df46d8555d"
    roomId: "77fea156-c7a3-05c2-8bd7-309f192a7514"
    agentId: "b850bc30-45f8-0041-a00a-83df46d8555d"
    error: "insert or update on table \"participants\" violates foreign key constraint \"participants_roomId_rooms_id_fk\""
```

### Root Cause
The foreign key constraint `participants_roomId_rooms_id_fk` enforces referential integrity between the `participants` and `rooms` tables. This constraint requires that any `roomId` in the `participants` table must reference an existing `id` in the `rooms` table.

The error occurred because the code was trying to insert participant records with a `roomId` that didn't exist yet in the `rooms` table.

### When This Issue Occurs

### Solution
Reordered the operations in `ensureConnections` to respect the foreign key constraint:

1. **First**: Create/ensure the room exists in the `rooms` table
2. **Then**: Create participants that reference that room

/**
 * Polls the comfy.icu API for workflow completion
 */
async function pollForCompletion(
  runtime: IAgentRuntime,
  workflowId: string,
  runId: string
): Promise<string | null> {
  const apiKey = getApiKey(runtime);
  const maxAttempts = 30; // 5 minutes with 10-second intervals
  let attempts = 0;

  while (attempts < maxAttempts) {
    try {
      const response = await fetch(
        `https://comfy.icu/api/v1/workflows/${workflowId}/runs/${runId}`,
        {
          headers: {
            Authorization: `Bearer ${apiKey}`,
          },
        }
      );

      if (!response.ok) {
        logger.warn(`Status check failed: ${response.status}`);
        await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
        attempts++;
        continue;
      }

      const status = (await response.json()) as {
        status: "STARTED" | "COMPLETED" | "ERROR";
        output?: Array<{ url: string }>;
        error?: string;
      };

      if (
        status.status === "COMPLETED" &&
        status.output &&
        status.output.length > 0
      ) {
        // Get the first image URL
        const imageUrl = status.output[0].url;
        if (imageUrl) {
          logger.info(`Workflow completed, image URL: ${imageUrl}`);
          return imageUrl;
        } else {
          logger.error("Workflow completed but no image URL found");
          return null;
        }
      } else if (status.status === "ERROR") {
        logger.error(`Workflow failed: ${status.error || "Unknown error"}`);
        return null;
      }

      // Still processing, wait and try again
      logger.debug(`Workflow status: ${status.status}, waiting...`);
      await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
      attempts++;
    } catch (error) {
      const message = error instanceof Error ? error.message : String(error);
      logger.error(`Error checking workflow status: ${message}`);
      await new Promise((resolve) => setTimeout(resolve, 10000)); // Wait 10 seconds
      attempts++;
    }
  }

  logger.error("Timed out waiting for workflow completion");
  return null;
}

### Problem
The `ensureConnections` function was attempting to insert participants before ensuring the room exists, causing a foreign key constraint violation:

```
[2025-06-04 22:42:47] ERROR: Error adding participants
    entityIdSample: "b850bc30-45f8-0041-a00a-83df46d8555d"
    roomId: "77fea156-c7a3-05c2-8bd7-309f192a7514"
    agentId: "b850bc30-45f8-0041-a00a-83df46d8555d"
    error: "insert or update on table \"participants\" violates foreign key constraint \"participants_roomId_rooms_id_fk\""
```

### Root Cause
The foreign key constraint `participants_roomId_rooms_id_fk` enforces referential integrity between the `participants` and `rooms` tables. This constraint requires that any `roomId` in the `participants` table must reference an existing `id` in the `rooms` table.

The error occurred because the code was trying to insert participant records with a `roomId` that didn't exist yet in the `rooms` table.

### When This Issue Occurs
This is most obvious when having an agent participate in multiple Discord channels on the same server, where Discord channels map to rooms in the database. The race condition happens when the agent tries to join multiple channels simultaneously before the room records are properly created.

### Solution
Reordered the operations in `ensureConnections` to respect the foreign key constraint:

1. **First**: Create/ensure the room exists in the `rooms` table
2. **Then**: Create participants that reference that room
